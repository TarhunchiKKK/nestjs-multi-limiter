# Async Configuration

Here are examples of `RateLimiterModule` async configuration.

## Factory configuration

```typescript title="app.module.ts"
RateLimiterModule.forRootAsync({
    imports: [ConfigModule, RedisModule],
    inject: [ConfigService, RedisService],
    useFactory: (configService: ConfigService, redisService: RedisService) => ({
        scope: this.configService.getOrThrow<string>("RATE_LIMIT_SCOPE"),
        storage: {
            type: "redis",
            gcTime: redisService
        },
        strategy: this.configService.getOrThrow<Strategies>("RATE_LIMIT_STRATEGY"),
        strategyOptions: {
            fixedWindow: {/* */},
            tokenBucket: {/* */},
            slidingWindowCounter: {/* */},
            slidingWindowLog: {/* */},
            leakyBucket: {/* */}
        },
        defaultProviders: {
            keyExtractor: MyCustomKeyExtractor,
            errorFactory: MyCustomErrorFactory,
            optionsFactory: MyCustomOptionsFactory
        }
    })
})
```

## Class configuration

Setup provider for configuration:

```typescript title="rate-limiter-config.service.ts"
import { Injectable } from "@nestjs/common";
import type { IRateLimiterModuleOptionsFactory, RateLimiterModuleOptions } from "nestjs-multi-limiter";

@Injectable()
export class RateLimiterConfigService implements IRateLimiterModuleOptionsFactory {
    public createRateLimiterModuleOptions(): RateLimiterModuleOptions | Promise<RateLimiterModuleOptions> {
        return {
            storage: {
                type: "in-memory"
                // other fields...
            }
        }
    }
}
```

Use this provider to configure `RateLimiterModule`:

```typescript title="app.module.ts"
RateLimiterModule.forRootAsync({
    useClass: RateLimiterConfigService
})
```

## Configuration with existing provider

Setup provider for configuration:

```typescript title="rate-limiter-config.service.ts"
import { Injectable } from "@nestjs/common";
import type { IRateLimiterModuleOptionsFactory, RateLimiterModuleOptions } from "nestjs-multi-limiter";

@Injectable()
export class RateLimiterConfigService implements IRateLimiterModuleOptionsFactory {
    public createRateLimiterModuleOptions(): RateLimiterModuleOptions | Promise<RateLimiterModuleOptions> {
        return {
            storage: {
                type: "in-memory"
                // other fields...
            }
        }
    }
}
```

Setup module with configuration provider:

```typescript title="app-config.module.ts"
@Module({
    providers: [RateLimiterConfigService],
    exports: [RateLimiterConfigService]
})
export class AppConfigModule {}
```

Use existing provider to configure `RateLimiterModule`:

```typescript title="app.module.ts"
RateLimiterModule.forRootAsync({
    imports: [AppConfigModule],
    useExisting: RateLimiterConfigService
})
```
